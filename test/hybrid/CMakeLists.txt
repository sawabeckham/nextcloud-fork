cmake_minimum_required(VERSION 3.16)
project(NextcloudForkPolicyTests LANGUAGES CXX)
if(APPLE)
    enable_language(OBJCXX)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
foreach(mode IN ITEMS default hybrid)
    add_executable(HybridSyncPolicy_${mode} ../testhybridsyncpolicy.cpp)
    target_include_directories(HybridSyncPolicy_${mode} PRIVATE ../../src)
    if(mode STREQUAL "hybrid")
        target_compile_definitions(HybridSyncPolicy_${mode} PRIVATE NEXTCLOUD_EXPERIMENTAL_HYBRID_SYNC)
    endif()
    add_test(NAME HybridSyncPolicy_${mode} COMMAND HybridSyncPolicy_${mode})
endforeach()
if(APPLE)
    add_executable(HybridSyncPathsTest ../testhybridsyncpaths.cpp)
    target_include_directories(HybridSyncPathsTest PRIVATE ../../src)
    add_test(NAME HybridSyncPathsTest COMMAND HybridSyncPathsTest)
endif()
add_test(NAME ForkBranding COMMAND ${CMAKE_COMMAND} -P ${CMAKE_CURRENT_SOURCE_DIR}/branding.cmake)
add_test(NAME MacRuntimeStaging COMMAND ${CMAKE_COMMAND} -P ${CMAKE_CURRENT_SOURCE_DIR}/staging.cmake)
if(APPLE)
    find_program(SWIFT_EXECUTABLE swift REQUIRED)
    add_test(NAME ForkPalette COMMAND ${SWIFT_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../admin/fork/make-fork-icons.swift --test-palette)
    set_tests_properties(ForkPalette PROPERTIES ENVIRONMENT
        "CLANG_MODULE_CACHE_PATH=${CMAKE_CURRENT_BINARY_DIR}/clang-cache;SWIFT_MODULECACHE_PATH=${CMAKE_CURRENT_BINARY_DIR}/swift-cache")
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME ForkFileProviderLocationProfile COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_file_provider_location_profile.py)
    add_test(NAME ForkInstaller COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_installer.py)
    add_test(NAME ForkInstallerPublication COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_publish_installer.py)
endif()

if(APPLE)
    find_package(Qt6 COMPONENTS Core QUIET)
    if(Qt6Core_FOUND)
        add_executable(HybridSyncQtPathsTest qtpaths.cpp ../../src/common/utility_mac_sandbox.mm)
        target_include_directories(HybridSyncQtPathsTest PRIVATE ../../src)
        set_source_files_properties(../../src/common/utility_mac_sandbox.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
        target_link_libraries(HybridSyncQtPathsTest PRIVATE Qt6::Core "-framework Foundation")
        add_test(NAME HybridSyncQtPathsTest COMMAND HybridSyncQtPathsTest)

        add_executable(ForkDomainDisplayNameTest domainname.cpp)
        target_include_directories(ForkDomainDisplayNameTest PRIVATE ../../src)
        target_link_libraries(ForkDomainDisplayNameTest PRIVATE Qt6::Core)
        add_test(NAME ForkDomainDisplayNameTest COMMAND ForkDomainDisplayNameTest)
    endif()
endif()
